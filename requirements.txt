  <script>
    const form = document.getElementById("form");
    const statusEl = document.getElementById("status");
    const buttons = document.querySelectorAll("button");

    function setButtons(disabled) {
      buttons.forEach(b => b.disabled = disabled);
    }

    function appendIfExists(fd, name) {
      const input = form.querySelector(`[name="${name}"]`);
      if (input && input.files.length > 0) {
        fd.append(name, input.files[0]);
      }
    }

    async function enviar(fd, etiqueta) {
      setButtons(true);
      statusEl.textContent = `Procesando ${etiqueta}...`;

      try {
        const res = await fetch("/conciliar", { method: "POST", body: fd });

        if (!res.ok) {
          let msg = "Error procesando la conciliación.";
          try {
            const data = await res.json();
            msg = data.error || msg;
          } catch {}
          throw new Error(msg);
        }

        const blob = await res.blob();
        const url = URL.createObjectURL(blob);
        const a = document.createElement("a");
        a.href = url;
        a.download = `CONCILIACIONES_BROU_${etiqueta.toUpperCase()}.zip`;
        document.body.appendChild(a);
        a.click();
        a.remove();
        URL.revokeObjectURL(url);

        statusEl.textContent = `Listo. ZIP descargado para ${etiqueta}.`;
      } catch (err) {
        statusEl.textContent = err.message;
      } finally {
        setButtons(false);
      }
    }

    function conciliarSolo(moneda) {
      const fd = new FormData();
      appendIfExists(fd, `sap_${moneda}`);
      appendIfExists(fd, `banco_${moneda}`);
      enviar(fd, moneda);
    }

    function conciliarTodo() {
      const fd = new FormData();
      ["usd", "uyu", "eur"].forEach(m => {
        appendIfExists(fd, `sap_${m}`);
        appendIfExists(fd, `banco_${m}`);
      });
      enviar(fd, "todo");
    }
  </script>
